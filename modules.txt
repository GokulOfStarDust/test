Program:3 Design a program to simulate 4 Queen Problem

def solve_n_queens(n):
    def is_safe(board, row, col):
        for i in range(row):
            if board[i] == col or abs(board[i] - col) == abs(i - row):
                return False
        return True
 
    def backtrack(row, board):
        if row == n:
            result.append(board[:])
            return
        for col in range(n):
            if is_safe(board, row, col):
                board[row] = col
                backtrack(row + 1, board)
 
    result = []
    backtrack(0, [-1] * n)
    return result
solutions = solve_n_queens(4)
for sol in solutions:
    for i in sol:
        print("." * i + "Q" + "." * (len(sol) - i - 1))
    print()

def hanoi(n,src,dst,aux):
    if n==1:
        print(f"Move Disk 1 from {src} to {dst}")
        return
    hanoi(n-1,src,aux,dst)
    print(f"Move Disk {n} from {src} to {dst}")
    hanoi(n-1,aux,dst,src)

Program:5 Create a program to implement Hill Climbing Problem.

import random,matplotlib.pyplot as plt,numpy as np
def f(x):
    return -(x-3)**2+9
def hill_climb(iters=1000,step=0.1):
    x,path=random.uniform(-10,10),[]
    for _ in range(iters):
        nx=x+random.uniform(-step,step)
        if f(nx)>f(x):x=nx
        path.append(x)
    return x,f(x),path
best_x,best_val,path=hill_climb()
x=np.linspace(-10,10,400)
plt.plot(x,f(x),'b-',label="f(x)")
plt.plot(path,[f(p) for p in path],'ro',label="path")
plt.scatter(best_x,best_val,c="g",s=80,label="best")
plt.legend()
plt.show()

Program:6 Design a program to implement A* algorithm.

import heapq
def astar(start,goal,graph,heuristic):
    open_list=[(heuristic[start],0,start,[start])]
    visited=set()
    while open_list:
        f,g,node,path=heapq.heappop(open_list)
        if node==goal:
            return path,g
        if node in visited:
            continue
        visited.add(node)
        for neighbour,cost in graph[node]:
            if neighbour not in visited:
                g_new=g+cost
                f_new=g_new+heuristic[neighbour]
                heapq.heappush(open_list,(f_new,g_new,neighbour,path+[neighbour]))
    return none,float("inf")
graph={
    'A':[('B',1),('C',3)], 'B':[('D',1),('E',5)],'C':[('F',2)],'D':[('G',2)],'E':[('G',1)],'F':[('G',2)],'G':[]}
heuristic={'A':7,'B':6,'C':5,'D':4,'E':2,'F':1,'G':0}
path,cost=astar('A','G',graph,heuristic)
print("Shortest Path:",path)
print("Total Cost:",cost)

Program:16 Demonstrate Sentiment Analysis with Naïve Bayes classifier.
import pandas as pd
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.model_selection import train_test_split
from sklearn.naive_bayes import MultinomialNB
from sklearn.metrics import accuracy_score, classification_report

df = pd.read_csv('Z:/program_16_csv.csv')
vectorizer = CountVectorizer()
x = vectorizer.fit_transform(df['text'])
y = df['label']
x_train,x_test,y_train,y_test = train_test_split(x,y,test_size = 0.25,random_state = 42, stratify = y)
model = MultinomialNB()
model.fit(x_train, y_train)
y_pred = model.predict(x_test)
print("Accuracy :",accuracy_score(y_test,y_pred))
print(f"\n Classification Report:\n:",classification_report(y_test,y_pred))
new_texts = ["I really enjoyed this!","It was a horrible experience."]
new_features = vectorizer.transform(new_texts)
predictions = model.predict(new_features)
for text,sentiment in zip(new_texts,predictions):
    print(f"Text : '{text}'->Sentiment:{sentiment}")
